    75→export default function HomePage() {
    76→  // Auth state
    77→  const [user, setUser] = useState<User | null>(null)
    78→  const [loginEmail, setLoginEmail] = useState('')
    79→  const [loginPassword, setLoginPassword] = useState('')
    80→  const [loginError, setLoginError] = useState('')
    81→  const [loginLoading, setLoginLoading] = useState(false)
    82→
    83→  // Products state
    84→  const [products, setProducts] = useState<Product[]>([])
    85→  const [categories, setCategories] = useState<CategoryInfo[]>([])
    86→  const [totalCount, setTotalCount] = useState(0)
    87→  const [archivedCount, setArchivedCount] = useState(0)
    88→  const [searchQuery, setSearchQuery] = useState('')
    89→  const [selectedCategory, setSelectedCategory] = useState('Todos')
    90→  const [showArchived, setShowArchived] = useState(false)
    91→  const [productsLoading, setProductsLoading] = useState(false)
    92→
    93→  // Modal state
    94→  const [modalOpen, setModalOpen] = useState(false)
    95→  const [editingProduct, setEditingProduct] = useState<Product | null>(null)
    96→  const [form, setForm] = useState<ProductForm>(emptyForm)
    97→  const [saving, setSaving] = useState(false)
    98→
    99→  // Check for existing session on mount
   100→  useEffect(() => {
   101→    const stored = localStorage.getItem('padaria_user')
   102→    if (stored) {
   103→      try {
   104→        setUser(JSON.parse(stored))
   105→      } catch {
   106→        localStorage.removeItem('padaria_user')
   107→      }
   108→    }
   109→  }, [])
   110→
   111→  // Fetch products when user is authenticated or filters change
   112→  const fetchProducts = useCallback(async () => {
   113→    if (!user) return
   114→    setProductsLoading(true)
   115→    try {
   116→      const params = new URLSearchParams()
   117→      if (searchQuery) params.set('search', searchQuery)
   118→      if (selectedCategory !== 'Todos') params.set('categoria', selectedCategory)
   119→      params.set('archived', showArchived ? 'true' : 'false')
   120→
   121→      const res = await fetch(`/api/products?${params.toString()}`)
   122→      const data = await res.json()
   123→      setProducts(data.products)
   124→      setCategories(data.categories)
   125→      setTotalCount(data.total)
   126→      setArchivedCount(data.archivedCount || 0)
   127→    } catch (err) {
   128→      console.error('Error fetching products:', err)
   129→      toast.error('Erro ao carregar produtos')
   130→    } finally {
   131→      setProductsLoading(false)
   132→    }
   133→  }, [user, searchQuery, selectedCategory, showArchived])
   134→
   135→  useEffect(() => {
   136→    fetchProducts()
   137→  }, [fetchProducts])
   138→
   139→  // Login handler
   140→  const handleLogin = async (e: React.FormEvent) => {
   141→    e.preventDefault()
   142→    setLoginError('')
   143→    setLoginLoading(true)
   144→
   145→    try {
   146→      const res = await fetch('/api/auth/login', {
   147→        method: 'POST',
   148→        headers: { 'Content-Type': 'application/json' },
   149→        body: JSON.stringify({ email: loginEmail, password: loginPassword }),
   150→      })
   151→
   152→      const data = await res.json()
   153→
   154→      if (!res.ok) {
   155→        setLoginError(data.error || 'Erro ao fazer login')
   156→        return
   157→      }
   158→
   159→      setUser(data)
   160→      localStorage.setItem('padaria_user', JSON.stringify(data))
   161→      toast.success(`Bem-vindo, ${data.name}!`)
   162→    } catch {
   163→      setLoginError('Erro de conexão com o servidor')
   164→    } finally {
   165→      setLoginLoading(false)
   166→    }
   167→  }
   168→
   169→  // Archive/Unarchive product
   170→  const handleToggleArchive = async (product: Product) => {
   171→    try {
   172→      const res = await fetch(`/api/products/${product.id}`, {
   173→        method: 'PUT',
   174→        headers: {
   175→          'Content-Type': 'application/json',
   176→          'Authorization': `Bearer ${user?.id}`,
   177→        },
   178→        body: JSON.stringify({ archived: !product.archived }),
   179→      })
   180→      if (!res.ok) {
   181→        toast.error('Erro ao arquivar produto')
   182→        return
   183→      }
   184→      toast.success(product.archived ? 'Produto restaurado!' : 'Produto arquivado!')
   185→      fetchProducts()
   186→    } catch {
   187→      toast.error('Erro de conexão')
   188→    }
   189→  }
   190→
   191→  // Delete product permanently
   192→  const handleDelete = async (product: Product) => {
   193→    if (!confirm(`Excluir permanentemente "${product.descricao}"?`)) return
   194→    try {
   195→      const res = await fetch(`/api/products/${product.id}`, {
   196→        method: 'DELETE',
   197→        headers: { 'Authorization': `Bearer ${user?.id}` },
   198→      })
   199→      if (!res.ok) {
   200→        toast.error('Erro ao excluir produto')
   201→        return
   202→      }
   203→      toast.success('Produto excluído permanentemente!')
   204→      fetchProducts()
   205→    } catch {
   206→      toast.error('Erro de conexão')
   207→    }
   208→  }
   209→
   210→  // Logout handler
   211→  const handleLogout = () => {
   212→    setUser(null)
   213→    localStorage.removeItem('padaria_user')
   214→    setProducts([])
   215→    setCategories([])
   216→    setTotalCount(0)
   217→    setArchivedCount(0)
   218→    setSearchQuery('')
   219→    setSelectedCategory('Todos')
   220→    setShowArchived(false)
   221→    toast.info('Sessão encerrada')
   222→  }
   223→
   224→  // Open add modal
   225→  const handleAdd = () => {

   227→    setForm(emptyForm)
   228→    setModalOpen(true)
   229→  }
   230→

   232→  const handleEdit = (product: Product) => {
   233→    setEditingProduct(product)
   234→    setForm({
   235→      descricao: product.descricao,
   236→      codigoSistema: String(product.codigoSistema),
   237→      codigoBalanca: String(product.codigoBalanca),
   238→      unidade: product.unidade,
   239→      categoria: product.categoria,
   240→    })
   241→    setModalOpen(true)
   242→  }
   243→
   244→  // Save product (create or update)
   245→  const handleSave = async () => {
   246→    if (!form.descricao || !form.codigoSistema || !form.codigoBalanca) {
   247→      toast.error('Preencha todos os campos obrigatórios')
   248→      return
   249→    }
   250→
   251→    setSaving(true)
   252→    try {
   253→      const payload = {
   254→        descricao: form.descricao,
   255→        codigoSistema: parseInt(form.codigoSistema),
   256→        codigoBalanca: parseInt(form.codigoBalanca),
   257→        unidade: form.unidade,
   258→        categoria: form.categoria,
   259→      }
   260→
   261→      let res: Response
   262→      if (editingProduct) {
   263→        res = await fetch(`/api/products/${editingProduct.id}`, {
   264→          method: 'PUT',
   265→          headers: {
   266→            'Content-Type': 'application/json',
   267→            'Authorization': `Bearer ${user?.id}`,
   268→          },
   269→          body: JSON.stringify(payload),
   270→        })
   271→      } else {
   272→        res = await fetch('/api/products', {
   273→          method: 'POST',
   274→          headers: {
   275→            'Content-Type': 'application/json',
   276→            'Authorization': `Bearer ${user?.id}`,
   277→          },
   278→          body: JSON.stringify(payload),
   279→        })
   280→      }
   281→
   282→      if (!res.ok) {
   283→        const data = await res.json()
   284→        toast.error(data.error || 'Erro ao salvar produto')
   285→        return
   286→      }
   287→
   288→      toast.success(editingProduct ? 'Produto atualizado!' : 'Produto criado!')
   289→      setModalOpen(false)
   290→      fetchProducts()
   291→    } catch {
   292→      toast.error('Erro de conexão')
   293→    } finally {
   294→      setSaving(false)
   295→    }
   296→  }
   297→
   298→  // Get filtered count for category chip
   299→  const getCategoryCount = (catName: string) => {
   300→    if (catName === 'Todos') return totalCount
   301→    const found = categories.find((c) => c.name === catName)
   302→    return found?.count || 0
   303→  }
   304→
   305→  // All categories for display
   306→  const allCategories = ['Todos', ...CATEGORY_OPTIONS.filter((c) => categories.some((cat) => cat.name === c) || selectedCategory === c)]
   307→
   308→  // ==================== LOGIN SCREEN ====================
   309→  if (!user) {
   310→    return (
   311→      <div className="min-h-screen flex items-center justify-center p-4" style={{ backgroundColor: 'oklch(0.96 0.01 85)' }}>
   312→        <motion.div
   313→          initial={{ opacity: 0, y: 20 }}
   314→          animate={{ opacity: 1, y: 0 }}
   315→          transition={{ duration: 0.5 }}
   316→          className="w-full max-w-sm"
   317→        >
   318→          <Card className="border-0 shadow-xl">
   319→            <CardHeader className="text-center pb-2">
   320→              <div className="mx-auto mb-3 w-16 h-16 rounded-2xl flex items-center justify-center text-3xl" style={{ backgroundColor: 'oklch(0.68 0.15 55)' }}>
   321→                🍞
   322→              </div>
   323→              <CardTitle className="text-2xl font-bold" style={{ color: 'oklch(0.35 0.08 50)' }}>
   324→                Padaria TATICO
   325→              </CardTitle>
   326→              <CardDescription className="text-sm">
   327→                Sistema de Códigos de Produtos
   328→              </CardDescription>
   329→            </CardHeader>
   330→            <CardContent>
   331→              <form onSubmit={handleLogin} className="space-y-4">
   332→                <div className="space-y-2">
   333→                  <Label htmlFor="email">Email</Label>
   334→                  <Input
   335→                    id="email"
   336→                    type="email"
   337→                    placeholder="admin@padaria.com"
   338→                    value={loginEmail}
   339→                    onChange={(e) => setLoginEmail(e.target.value)}
   340→                    required
   341→                    className="h-11"
   342→                    autoComplete="email"
   343→                  />
   344→                </div>
   345→                <div className="space-y-2">
   346→                  <Label htmlFor="password">Senha</Label>
   347→                  <Input
   348→                    id="password"
   349→                    type="password"
   350→                    placeholder="••••••••"
   351→                    value={loginPassword}
   352→                    onChange={(e) => setLoginPassword(e.target.value)}
   353→                    required
   354→                    className="h-11"
   355→                    autoComplete="current-password"
   356→                  />
   357→                </div>
   358→                {loginError && (
   359→                  <motion.div
   360→                    initial={{ opacity: 0, height: 0 }}
   361→                    animate={{ opacity: 1, height: 'auto' }}
   362→                    className="text-sm text-destructive bg-destructive/10 p-2 rounded-md"
   363→                  >
   364→                    {loginError}
   365→                  </motion.div>
   366→                )}
   367→                <Button
   368→                  type="submit"
   369→                  className="w-full h-11 text-base font-semibold"
   370→                  disabled={loginLoading}
   371→                  style={{ backgroundColor: 'oklch(0.68 0.15 55)' }}
   372→                >
   373→                  {loginLoading ? (
   374→                    <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   375→                  ) : null}
   376→                  {loginLoading ? 'Entrando...' : 'Entrar'}
   377→                </Button>
   378→              </form>
   379→            </CardContent>
   380→          </Card>
   381→        </motion.div>
   382→      </div>
   383→    )
   384→  }
   385→
   386→  // ==================== MAIN APP ====================
   387→  return (
   388→    <div className="min-h-screen flex flex-col" style={{ backgroundColor: 'oklch(0.96 0.01 85)' }}>
   389→      {/* Header */}
   390→      <header className="sticky top-0 z-40 border-b" style={{ backgroundColor: 'oklch(0.68 0.15 55)', borderColor: 'oklch(0.58 0.12 55)' }}>
   391→        <div className="max-w-4xl mx-auto px-4 py-3 flex items-center justify-between">
   392→          <div className="flex items-center gap-2">
   393→            <span className="text-2xl">🍞</span>
   394→            <h1 className="text-lg font-bold text-white tracking-tight">Padaria TATICO</h1>
   395→          </div>
   396→          <div className="flex items-center gap-3">
   397→            <span className="text-sm text-white/80 hidden sm:inline">
   398→              {user.name}
   399→            </span>
   400→            <Button
   401→              variant="ghost"
   402→              size="sm"
   403→              onClick={handleLogout}
   404→              className="text-white/90 hover:text-white hover:bg-white/20 h-9 px-2"
   405→            >
   406→              <LogOut className="h-4 w-4 mr-1" />
   407→              <span className="hidden sm:inline">Sair</span>
   408→            </Button>
   409→          </div>
   410→        </div>
   411→      </header>
   412→
   413→      {/* Search Bar */}
   414→      <div className="sticky top-[52px] z-30 border-b" style={{ backgroundColor: 'oklch(0.96 0.01 85)', borderColor: 'oklch(0.88 0.03 85)' }}>
   415→        <div className="max-w-4xl mx-auto px-4 py-3">
   416→          <div className="relative">
   417→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   418→            <Input
   419→              placeholder="Buscar por descrição, código sistema ou balança..."
   420→              value={searchQuery}
   421→              onChange={(e) => setSearchQuery(e.target.value)}
   422→              className="pl-9 h-11 bg-white/80 border-0 shadow-sm focus-visible:ring-2"
   423→              style={{ '--tw-ring-color': 'oklch(0.68 0.15 55)' } as React.CSSProperties}
   424→            />
   425→            {searchQuery && (
   426→              <button
   427→                onClick={() => setSearchQuery('')}
   428→                className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   429→              >
   430→                <X className="h-4 w-4" />
   431→              </button>
   432→            )}
   433→          </div>
   434→        </div>
   435→      </div>
   436→
   437→      {/* Category Filter Chips */}
   438→      <div className="sticky top-[108px] z-20 border-b" style={{ backgroundColor: 'oklch(0.96 0.01 85)', borderColor: 'oklch(0.88 0.03 85)' }}>
   439→        <div className="max-w-4xl mx-auto">
   440→          <div className="flex gap-2 px-4 py-3 overflow-x-auto hide-scrollbar">
   441→            {/* Archived toggle */}
   442→            <button
   443→              onClick={() => { setShowArchived(!showArchived); setSelectedCategory('Todos') }}
   444→              className={`
   445→                flex-shrink-0 inline-flex items-center gap-1.5 px-3 py-2 rounded-full text-sm font-medium
   446→                transition-all duration-200 whitespace-nowrap
   447→                ${showArchived
   448→                  ? 'text-white shadow-md scale-105'
   449→                  : 'bg-white/60 text-foreground/70 hover:bg-white/90 hover:text-foreground shadow-sm'

   452→              style={showArchived ? { backgroundColor: 'oklch(0.55 0.08 250)' } : undefined}

   457→                ({archivedCount})
   458→              </span>
   459→            </button>
   460→
   461→            {!showArchived && allCategories.map((cat) => {
   462→              const isActive = selectedCategory === cat
   463→              const emoji = cat === 'Todos' ? '📋' : (CATEGORY_EMOJI[cat] || '📦')
   464→              const count = getCategoryCount(cat)
   465→              const displayName = cat === 'Produtos Amanhecidos' ? 'Amanhecidos' : cat
   466→
   467→              return (
   468→                <button
   469→                  key={cat}
   470→                  onClick={() => setSelectedCategory(cat)}

   473→                    transition-all duration-200 whitespace-nowrap

   475→                      ? 'text-white shadow-md scale-105'

   479→                  style={isActive ? { backgroundColor: 'oklch(0.68 0.15 55)' } : undefined}

   481→                  <span>{emoji}</span>
   482→                  <span>{displayName}</span>

   490→        </div>
   491→      </div>
   492→
   493→      {/* Product List */}
   494→      <main className="flex-1 max-w-4xl mx-auto w-full px-4 py-4 pb-24">
   495→        {productsLoading ? (
   496→          <div className="flex items-center justify-center py-12">

   498→          </div>
   499→        ) : products.length === 0 ? (
   500→          <div className="text-center py-12 text-muted-foreground">
   501→            <Package className="h-12 w-12 mx-auto mb-3 opacity-30" />

   522→                          <h3 className="font-semibold text-sm leading-tight mb-1.5 truncate" title={product.descricao}>

   528→                              className="text-xs font-mono px-2"
